cmake_minimum_required(VERSION 3.22)
project(Kavrune VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(KAVRUNE_BUILD_DESKTOP "Build the GTK4 desktop application" ON)
option(KAVRUNE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
include(GNUInstallDirs)
include(CTest)
find_package(Threads REQUIRED)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0 gio-unix-2.0)
add_library(kavrune-core src/core.cpp)
target_include_directories(kavrune-core PUBLIC include)
target_link_libraries(kavrune-core PUBLIC OpenSSL::Crypto Threads::Threads)
add_library(kavrune-storage src/storage.cpp)
target_link_libraries(kavrune-storage PUBLIC kavrune-core PkgConfig::GIO)
foreach(target kavrune-core kavrune-storage)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wformat=2 -fstack-protector-strong)
endforeach()
if(KAVRUNE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(kavrune-core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(kavrune-storage PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
add_executable(kavrune-cli src/cli.cpp)
target_link_libraries(kavrune-cli PRIVATE kavrune-storage)
target_compile_options(kavrune-cli PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
if(KAVRUNE_BUILD_DESKTOP)
  pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4>=4.12 libadwaita-1>=1.4)
  find_program(GLIB_COMPILE_RESOURCES glib-compile-resources REQUIRED)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/resources.c
    COMMAND ${GLIB_COMPILE_RESOURCES} ${CMAKE_CURRENT_SOURCE_DIR}/data/resources.xml
            --sourcedir=${CMAKE_CURRENT_SOURCE_DIR}/data --generate-source
            --target=${CMAKE_CURRENT_BINARY_DIR}/resources.c
    DEPENDS data/resources.xml data/style.css data/icon.svg VERBATIM)
  add_executable(kavrune src/desktop.cpp ${CMAKE_CURRENT_BINARY_DIR}/resources.c)
  target_link_libraries(kavrune PRIVATE kavrune-storage PkgConfig::GTK)
  target_compile_definitions(kavrune PRIVATE GETTEXT_PACKAGE="kavrune" KAVRUNE_VERSION="${PROJECT_VERSION}" KAVRUNE_LOCALE_DIR="${CMAKE_INSTALL_FULL_LOCALEDIR}")
  target_compile_options(kavrune PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
  install(TARGETS kavrune RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES data/org.kavrune.Kavrune.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES data/icon.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME org.kavrune.Kavrune.svg)
  install(FILES data/org.kavrune.Kavrune.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
endif()
install(TARGETS kavrune-cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE NOTICE README.md SECURITY.md DISCLAIMER.md CONTRIBUTING.md
  CODE_OF_CONDUCT.md SUPPORT.md CHANGELOG.md THIRD_PARTY_LICENSES.md
  DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/kavrune)
install(DIRECTORY docs DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/kavrune)
if(BUILD_TESTING)
  add_executable(core-tests tests/core_tests.cpp)
  target_link_libraries(core-tests PRIVATE kavrune-core)
  add_test(NAME core-safety COMMAND core-tests)
  add_executable(storage-tests tests/storage_tests.cpp)
  target_link_libraries(storage-tests PRIVATE kavrune-core PkgConfig::GIO)
  add_test(NAME storage-safety COMMAND storage-tests)
  add_test(NAME cli-help COMMAND kavrune-cli --help)
  add_test(NAME cli-safe-self-test COMMAND kavrune-cli self-test --json)
  add_test(NAME cli-file-simulation COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test-cli.sh $<TARGET_FILE:kavrune-cli>)
  find_program(KAVRUNE_QEMU_EXECUTABLE NAMES qemu-system-x86_64 PATHS /usr/bin /usr/local/bin NO_DEFAULT_PATH)
  if(KAVRUNE_QEMU_EXECUTABLE)
    add_test(NAME synthetic-bios-boot COMMAND ${CMAKE_COMMAND} -E env QEMU=${KAVRUNE_QEMU_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test-boot.sh $<TARGET_FILE:kavrune-cli>)
    set_tests_properties(synthetic-bios-boot PROPERTIES TIMEOUT 20 SKIP_RETURN_CODE 77)
  endif()
endif()
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME kavrune)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "Kavrune contributors (see SECURITY.md)")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native Linux USB image writer with readback verification")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libc6 (>= 2.35), libstdc++6 (>= 12), libgtk-4-1 (>= 4.12), libadwaita-1-0 (>= 1.4), libglib2.0-0t64, libssl3t64, udisks2 (>= 2.9), polkitd, dosfstools, exfatprogs, e2fsprogs")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
include(CPack)
