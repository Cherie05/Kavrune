cmake_minimum_required(VERSION 3.22)
project(Kavrune VERSION 0.1.0 LANGUAGES C CXX)
configure_file(include/kavrune/version.hpp.in generated/kavrune/version.hpp @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CheckPIESupported)
check_pie_supported(LANGUAGES C CXX)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_link_options(-Wl,-z,relro,-z,now,-z,noexecstack)
endif()
option(KAVRUNE_BUILD_DESKTOP "Build the GTK4 desktop application" ON)
option(KAVRUNE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
include(GNUInstallDirs)
include(CTest)
find_package(Threads REQUIRED)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0 gio-unix-2.0)
add_library(kavrune-core src/core.cpp)
target_include_directories(kavrune-core PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(kavrune-core PUBLIC OpenSSL::Crypto Threads::Threads)
add_library(kavrune-storage src/storage.cpp)
target_link_libraries(kavrune-storage PUBLIC kavrune-core PkgConfig::GIO)
foreach(target kavrune-core kavrune-storage)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wformat=2 -fstack-protector-strong)
endforeach()
if(KAVRUNE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(kavrune-core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(kavrune-storage PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
add_executable(kavrune-cli src/cli.cpp)
target_link_libraries(kavrune-cli PRIVATE kavrune-storage)
target_compile_options(kavrune-cli PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
if(KAVRUNE_BUILD_DESKTOP)
  pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4>=4.12 libadwaita-1>=1.4)
  find_program(GLIB_COMPILE_RESOURCES glib-compile-resources REQUIRED)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/resources.c
    COMMAND ${GLIB_COMPILE_RESOURCES} ${CMAKE_CURRENT_SOURCE_DIR}/data/resources.xml
            --sourcedir=${CMAKE_CURRENT_SOURCE_DIR}/data --generate-source
            --target=${CMAKE_CURRENT_BINARY_DIR}/resources.c
    DEPENDS data/resources.xml data/style.css data/icon.svg TERMS.md PRIVACY.md DISCLAIMER.md LICENSE THIRD_PARTY_LICENSES.md SUPPORT.md VERBATIM)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    # GResource generates a single byte string containing the offline notices.
    set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/resources.c PROPERTIES COMPILE_OPTIONS -Wno-overlength-strings)
  endif()
  add_executable(kavrune src/desktop.cpp ${CMAKE_CURRENT_BINARY_DIR}/resources.c)
  target_link_libraries(kavrune PRIVATE kavrune-storage PkgConfig::GTK)
  target_compile_definitions(kavrune PRIVATE GETTEXT_PACKAGE="kavrune" KAVRUNE_LOCALE_DIR="${CMAKE_INSTALL_FULL_LOCALEDIR}")
  target_compile_options(kavrune PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
  install(TARGETS kavrune RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES data/org.kavrune.Kavrune.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES data/icon.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME org.kavrune.Kavrune.svg)
  install(FILES data/org.kavrune.Kavrune.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
endif()
install(TARGETS kavrune-cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE NOTICE README.md INSTALL.md SECURITY.md TERMS.md PRIVACY.md DISCLAIMER.md CONTRIBUTING.md
  CODE_OF_CONDUCT.md SUPPORT.md CHANGELOG.md THIRD_PARTY_LICENSES.md
  PRODUCT_REQUIREMENTS.md PRODUCT_STANDARDS.md ARCHITECTURE.md PRIVACY_AND_LEGAL.md TEST_PLAN.md RELEASE_CHECKLIST.md
  DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/kavrune)
install(DIRECTORY docs DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/kavrune)
if(BUILD_TESTING)
  find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
  add_executable(core-tests tests/core_tests.cpp)
  target_link_libraries(core-tests PRIVATE kavrune-core)
  add_test(NAME core-safety COMMAND core-tests)
  add_executable(storage-tests tests/storage_tests.cpp)
  target_link_libraries(storage-tests PRIVATE kavrune-core PkgConfig::GIO)
  add_test(NAME storage-safety COMMAND storage-tests)
  add_test(NAME cli-help COMMAND kavrune-cli --help)
  add_test(NAME cli-safe-self-test COMMAND kavrune-cli self-test --json)
  add_test(NAME cli-file-simulation COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test-cli.sh $<TARGET_FILE:kavrune-cli>)
  set(KAVRUNE_OFFLINE_DESKTOP_ARGS)
  if(KAVRUNE_BUILD_DESKTOP)
    add_executable(ui-selection-tests tests/ui_selection_tests.cpp)
    target_include_directories(ui-selection-tests PRIVATE include)
    target_link_libraries(ui-selection-tests PRIVATE PkgConfig::GTK)
    target_compile_options(ui-selection-tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wformat=2)
    add_executable(ui-workflow-tests tests/ui_workflow_tests.cpp ${CMAKE_CURRENT_BINARY_DIR}/resources.c)
    target_link_libraries(ui-workflow-tests PRIVATE kavrune-storage PkgConfig::GTK)
    target_compile_definitions(ui-workflow-tests PRIVATE GETTEXT_PACKAGE="kavrune" KAVRUNE_LOCALE_DIR="${CMAKE_INSTALL_FULL_LOCALEDIR}")
    target_compile_options(ui-workflow-tests PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
    find_program(KAVRUNE_XVFB_RUN NAMES xvfb-run)
    find_program(KAVRUNE_XVFB NAMES Xvfb)
    find_program(KAVRUNE_DBUS_RUN_SESSION NAMES dbus-run-session)
    if(NOT KAVRUNE_SANITIZERS)
      set(KAVRUNE_OFFLINE_DESKTOP_ARGS --desktop $<TARGET_FILE:kavrune>)
    endif()
    if(KAVRUNE_XVFB_RUN AND KAVRUNE_XVFB AND KAVRUNE_DBUS_RUN_SESSION)
      add_test(NAME gtk-drive-selection COMMAND ${KAVRUNE_DBUS_RUN_SESSION} -- ${KAVRUNE_XVFB_RUN} -a $<TARGET_FILE:ui-selection-tests>)
      add_test(NAME gtk-workflow COMMAND ${KAVRUNE_DBUS_RUN_SESSION} -- ${KAVRUNE_XVFB_RUN} -a $<TARGET_FILE:ui-workflow-tests>)
    elseif(KAVRUNE_XVFB_RUN AND KAVRUNE_XVFB)
      add_test(NAME gtk-drive-selection COMMAND ${KAVRUNE_XVFB_RUN} -a $<TARGET_FILE:ui-selection-tests>)
      add_test(NAME gtk-workflow COMMAND ${KAVRUNE_XVFB_RUN} -a $<TARGET_FILE:ui-workflow-tests>)
    else()
      add_test(NAME gtk-drive-selection COMMAND ui-selection-tests)
      add_test(NAME gtk-workflow COMMAND ui-workflow-tests)
    endif()
    set_tests_properties(gtk-drive-selection gtk-workflow PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77 ENVIRONMENT "GTK_A11Y=none;GSETTINGS_BACKEND=memory")
    # GTK's test backend retains accessibility relations for actual-control checks.
    set_property(TEST gtk-workflow PROPERTY ENVIRONMENT "GTK_A11Y=test;GSETTINGS_BACKEND=memory")
    if(KAVRUNE_XVFB_RUN AND KAVRUNE_XVFB)
      # xvfb-run locates its server through PATH, even when the helper itself
      # was found by an absolute path. Force GTK onto this private X11 server.
      get_filename_component(KAVRUNE_XVFB_DIR "${KAVRUNE_XVFB}" DIRECTORY)
      set_property(TEST gtk-drive-selection gtk-workflow APPEND PROPERTY ENVIRONMENT "GDK_BACKEND=x11")
      set_property(TEST gtk-drive-selection gtk-workflow PROPERTY ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${KAVRUNE_XVFB_DIR}")
    endif()
    if(KAVRUNE_SANITIZERS)
      # GTK retains process-wide caches; core/storage leak checks remain enabled.
      set_property(TEST gtk-drive-selection APPEND PROPERTY ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
      set_property(TEST gtk-workflow APPEND PROPERTY ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
    endif()
  endif()
  add_test(NAME offline-local-workflows COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/scripts/check-offline.py --cli $<TARGET_FILE:kavrune-cli>
    ${KAVRUNE_OFFLINE_DESKTOP_ARGS} --report ${CMAKE_CURRENT_BINARY_DIR}/offline-check.json)
  set_tests_properties(offline-local-workflows PROPERTIES TIMEOUT 90 SKIP_RETURN_CODE 77)
  if(KAVRUNE_XVFB)
    set_property(TEST offline-local-workflows PROPERTY ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${KAVRUNE_XVFB_DIR}")
  endif()
  find_program(KAVRUNE_QEMU_EXECUTABLE NAMES qemu-system-x86_64 PATHS /usr/bin /usr/local/bin NO_DEFAULT_PATH)
  if(KAVRUNE_QEMU_EXECUTABLE)
    add_test(NAME synthetic-bios-boot COMMAND ${CMAKE_COMMAND} -E env QEMU=${KAVRUNE_QEMU_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test-boot.sh $<TARGET_FILE:kavrune-cli>)
    set_tests_properties(synthetic-bios-boot PROPERTIES TIMEOUT 20 SKIP_RETURN_CODE 77)
  endif()
endif()
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME kavrune)
set(CPACK_DEBIAN_PACKAGE_SECTION utils)
set(CPACK_DEBIAN_PACKAGE_HOMEPAGE "https://github.com/Cherie05/Kavrune")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "Kavrune contributors (see SECURITY.md)")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native Linux USB image writer with readback verification")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libc6 (>= 2.35), libstdc++6 (>= 12), libgtk-4-1 (>= 4.12), libadwaita-1-0 (>= 1.4), libglib2.0-0t64, libssl3t64, udisks2 (>= 2.9), polkitd, dosfstools, exfatprogs, e2fsprogs")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
include(CPack)
